<script>
(function(){
  var s=document.querySelector('.ln-slider'); if(!s) return;
  var sl=s.querySelectorAll('.ln-slide'), dt=s.querySelectorAll('.ln-sl-dots button'); if(sl.length<2) return;
  s.querySelectorAll('img[loading="lazy"]').forEach(function(im){im.loading='eager';});
  var i=0, t=null, ms=6000;
  function go(n){
    sl[i].classList.remove('on'); sl[i].setAttribute('aria-hidden','true'); dt[i].classList.remove('on');
    i=(n+sl.length)%sl.length;
    for(var k=0;k<dt.length;k++){ dt[k].classList.toggle('done',k<i); }
    sl[i].classList.add('on'); sl[i].setAttribute('aria-hidden','false');
    void dt[i].offsetWidth; dt[i].classList.add('on'); plan();
  }
  function plan(){ clearTimeout(t); if(!s.classList.contains('paused')) t=setTimeout(function(){ go(i+1); },ms); }
  s.querySelector('.next').addEventListener('click',function(e){e.preventDefault();go(i+1);});
  s.querySelector('.prev').addEventListener('click',function(e){e.preventDefault();go(i-1);});
  dt.forEach(function(b,k){ b.addEventListener('click',function(e){e.preventDefault();go(k);}); });
  s.addEventListener('mouseenter',function(){ s.classList.add('paused'); clearTimeout(t); });
  s.addEventListener('mouseleave',function(){ s.classList.remove('paused'); go(i); });
  var x0=null;
  s.addEventListener('touchstart',function(e){ x0=e.touches[0].clientX; },{passive:true});
  s.addEventListener('touchend',function(e){ if(x0===null) return; var dx=e.changedTouches[0].clientX-x0; if(Math.abs(dx)>40) go(i+(dx<0?1:-1)); x0=null; });
  plan();
})();
</script>
